# VoiceOver), so send what the New Game screen's Start button sends.

wait_until 30000 TheApp.world ~= nil
wait 1500

screenshot 2_in_game
# So do the toolbar's tool buttons and the Hospital menu.

# Fax messages: the bottom panel is hidden, so the status pill shows each fax
# with the game's own icon, as many as the game shows (one of each kind; the
# rest wait behind them).
lua local function fax(kind, text, timeout) TheApp.ui.bottom_panel:queueMessage(kind, { { text = text }, choices = { { text = "OK", choice = "close" } } }, nil, timeout) end; fax("information", "first"); fax("report", "a report"); fax("disease", "a disease"); fax("personality", "a VIP", 2000); fax("information", "second"); fax("information", "third")
wait_until 15000 #corsixth.require("ios.json").decode(TheApp.native.toolbar_json).faxes == 4
lua local s = corsixth.require("ios.json").decode(TheApp.native.toolbar_json); local kinds = {}; for _, f in ipairs(s.faxes) do kinds[#kinds + 1] = f.kind .. (f.image and "+icon" or "") end; return table.concat(kinds, " ") .. ", waiting " .. s.faxes_waiting
expect corsixth.require("ios.json").decode(TheApp.native.toolbar_json).faxes_waiting == 2
expect corsixth.require("ios.json").decode(TheApp.native.toolbar_json).faxes[2].image ~= nil
# The VIP fax expires, so it shows how long is left.
expect corsixth.require("ios.json").decode(TheApp.native.toolbar_json).faxes[4].time_left ~= nil
wait 1000
screenshot 5_messages_waiting
# Tapping an icon opens that fax.
native_tap Report
wait_until 3000 TheApp.ui:getWindow(UIFax) ~= nil
expect TheApp.ui:getWindow(UIFax).icon.type == "report"
wait 500
screenshot 6_fax
