# The native main menu starts a new game; the native toolbar opens the build
# and hire windows.
controls direct
wait_until 30000 require("sdl").native.screen() == "main_menu"
wait 1000
screenshot 1_main_menu
# SwiftUI rows can't be tapped by scripts (no accessibility tree without
# VoiceOver), so send what the New Game screen's Start button sends.
lua require("sdl").native.post("new_game", '{"difficulty":"easy","tutorial":false}')
wait_until 30000 TheApp.world ~= nil
wait 1500
expect require("sdl").native.screen() == nil
screenshot 2_in_game
# Close the level's introduction (still Theme Hospital's own window).
expect TheApp.ui:getWindow(UIInformation) ~= nil
lua TheApp.ui:getWindow(UIInformation):close()
wait 300
native_tap Build Room
wait_until 3000 require("sdl").native.screen() == "room_catalog"
expect TheApp.ui:getWindow(UIBuildRoom) == nil
wait 500
screenshot 3_build_room
lua require("sdl").native.dismiss(); require("sdl").native.post("dismissed")
wait 500
native_tap Hire Staff
wait_until 3000 TheApp.ui:getWindow(UIHireStaff) ~= nil
wait 500
screenshot 4_hire_staff
lua TheApp.ui:getWindow(UIHireStaff):close()
