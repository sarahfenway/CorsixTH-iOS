# Save from the save window (tapping its text box and typing a name), then
# load the save back.
controls direct
wait_until 30000 TheApp.ui ~= nil
lua os.remove(TheApp.savegame_dir .. "AutoTest.sav")
lua TheApp:loadLevel(1, "easy")
wait_until 30000 TheApp.world ~= nil
wait 1000
lua TheApp.ui:getWindow(UIInformation):close()
lua TheApp.ui:addWindow(UISaveGame(TheApp.ui))
wait 500

expect not require("sdl").wm.isScreenKeyboardShown()
